function getMenuItems() {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('Menu_Master');
    if (!sheet || sheet.getLastRow() <= 1) {
      return getDefaultMenu();
    }
    
    var data = sheet.getDataRange().getValues();
    var items = [];
    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      if (row && row[0]) {
        items.push({
          ItemID: String(row[0]),
          ItemName: String(row[1] || ''),
          Category: String(row[2] || ''),
          Price: parseFloat(row[3]) || 0,
          Cost: parseFloat(row[4]) || 0,
          IsAvailable: row[5] === 'TRUE' || row[5] === true
        });
      }
    }
    return items.length > 0 ? items : getDefaultMenu();
  } catch (error) {
    return getDefaultMenu();
  }
}

function getMenuWithStock() {
  try {
    var menu = getMenuItems();
    var stock = getInventoryItems();
    
    var stockMap = {};
    for (var i = 0; i < stock.length; i++) {
      stockMap[stock[i].ItemID] = stock[i];
    }
    
    for (var i = 0; i < menu.length; i++) {
      if (stockMap[menu[i].ItemID]) {
        menu[i].Stock = stockMap[menu[i].ItemID].CurrentStock || 0;
        menu[i].MinStock = stockMap[menu[i].ItemID].MinStock || 0;
        menu[i].Unit = stockMap[menu[i].ItemID].Unit || '';
      } else {
        menu[i].Stock = 0;
        menu[i].MinStock = 0;
        menu[i].Unit = '';
      }
    }
    
    return menu;
  } catch (error) {
    return getMenuItems();
  }
}

function getTransactionHistory(limit) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('Transactions');
    
    if (!sheet) {
      sheet = ss.insertSheet('Transactions');
      var headers = [
        'TransactionID', 'Timestamp', 'CashierName', 'Items', 
        'Subtotal', 'Discount', 'Tax', 'Total', 
        'PaymentMethod', 'AmountPaid', 'Change', 'Status'
      ];
      sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
      sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold');
      createSampleTransactions();
      sheet = ss.getSheetByName('Transactions');
    }
    
    var lastRow = sheet.getLastRow();
    if (lastRow <= 1) {
      createSampleTransactions();
      lastRow = sheet.getLastRow();
    }
    
    if (lastRow <= 1) {
      return [];
    }
    
    var data = sheet.getDataRange().getValues();
    var transactions = [];
    var maxLimit = limit || 100;
    var count = 0;
    
    for (var i = data.length - 1; i >= 1 && count < maxLimit; i--) {
      var row = data[i];
      if (!row || !row[0] || row[0] === '' || row[0] === 'TransactionID') continue;
      
      var items = [];
      var itemsStr = String(row[3] || '[]');
      
      try {
        if (itemsStr && itemsStr !== '[]' && itemsStr !== '') {
          var cleanStr = itemsStr.replace(/\r/g, '').replace(/\n/g, '');
          
          try {
            var parsed = JSON.parse(cleanStr);
            if (Array.isArray(parsed)) {
              items = parsed;
            } else if (typeof parsed === 'object' && parsed !== null) {
              items = [parsed];
            }
          } catch (jsonError) {
            var itemMatches = itemsStr.match(/\{[^}]+\}/g);
            if (itemMatches) {
              for (var m = 0; m < itemMatches.length; m++) {
                var itemStr = itemMatches[m];
                var itemObj = {};
                
                var idMatch = itemStr.match(/["']?ItemID["']?\s*[:=]\s*["']?([^"',}]+)/i);
                if (idMatch) itemObj.ItemID = idMatch[1].trim();
                
                var nameMatch = itemStr.match(/["']?ItemName["']?\s*[:=]\s*["']?([^"',}]+)/i);
                if (nameMatch) itemObj.ItemName = nameMatch[1].trim();
                
                var qtyMatch = itemStr.match(/["']?Quantity["']?\s*[:=]\s*(\d+)/i);
                if (qtyMatch) itemObj.Quantity = parseInt(qtyMatch[1]);
                else itemObj.Quantity = 1;
                
                var priceMatch = itemStr.match(/["']?UnitPrice["']?\s*[:=]\s*(\d+)/i);
                if (priceMatch) itemObj.UnitPrice = parseFloat(priceMatch[1]);
                else itemObj.UnitPrice = 0;
                
                if (itemObj.ItemName) {
                  items.push(itemObj);
                }
              }
            }
            
            if (items.length === 0) {
              var nameMatch2 = itemsStr.match(/["']?ItemName["']?\s*[:=]\s*["']?([^"',}]+)/i);
              var qtyMatch2 = itemsStr.match(/["']?Quantity["']?\s*[:=]\s*(\d+)/i);
              var priceMatch2 = itemsStr.match(/["']?UnitPrice["']?\s*[:=]\s*(\d+)/i);
              
              if (nameMatch2) {
                items = [{
                  ItemName: nameMatch2[1].trim(),
                  Quantity: parseInt(qtyMatch2 ? qtyMatch2[1] : 1),
                  UnitPrice: parseFloat(priceMatch2 ? priceMatch2[1] : 0)
                }];
              }
            }
          }
        }
      } catch (e) {}
      
      if (items.length === 0 && row[3]) {
        var fallbackName = String(row[3]);
        var nameMatch = fallbackName.match(/["']?ItemName["']?\s*[:=]\s*["']?([^"',}]+)/i);
        if (nameMatch) {
          items = [{
            ItemName: nameMatch[1].trim(),
            Quantity: 1,
            UnitPrice: parseFloat(row[7]) / 1 || 0
          }];
        } else {
          items = [{
            ItemName: fallbackName.substring(0, 30) + '...',
            Quantity: 1,
            UnitPrice: parseFloat(row[7]) || 0
          }];
        }
      }
      
      var transaction = {
        TransactionID: String(row[0] || 'TRX-' + Date.now()),
        Timestamp: row[1] || new Date().toISOString(),
        CashierName: String(row[2] || 'Admin'),
        Items: items,
        Subtotal: parseFloat(row[4]) || 0,
        Discount: parseFloat(row[5]) || 0,
        Tax: parseFloat(row[6]) || 0,
        Total: parseFloat(row[7]) || 0,
        PaymentMethod: String(row[8] || 'Cash'),
        AmountPaid: parseFloat(row[9]) || 0,
        Change: parseFloat(row[10]) || 0,
        Status: String(row[11] || 'Completed')
      };
      
      if (transaction.TransactionID) {
        transactions.push(transaction);
        count++;
      }
    }
    
    return transactions;
  } catch (error) {
    return [];
  }
}


function debugTransactions() {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('Transactions');
    
    if (!sheet) {
      return { error: 'Sheet Transactions tidak ditemukan!' };
    }
    
    var lastRow = sheet.getLastRow();
    if (lastRow <= 1) {
      return { totalRows: 0, message: 'Tidak ada data transaksi' };
    }
    
    var data = sheet.getDataRange().getValues();
    var headers = data[0];
    var sample = [];
    for (var i = 1; i < Math.min(6, data.length); i++) {
      sample.push(data[i]);
    }
    
    return {
      totalRows: lastRow - 1,
      headers: headers,
      sampleData: sample,
      hasData: lastRow > 1
    };
  } catch (error) {
    return { error: error.message };
  }
}

function createSampleTransactions() {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('Transactions');
    if (!sheet) {
      sheet = ss.insertSheet('Transactions');
      sheet.appendRow(['TransactionID', 'Timestamp', 'CashierName', 'Items',
                       'Subtotal', 'Discount', 'Tax', 'Total',
                       'PaymentMethod', 'AmountPaid', 'Change', 'Status']);
    }

    var menuItems = getMenuItems();
    var cashiers = ['Admin', 'Budi Kasir'];
    var paymentMethods = ['Cash', 'QRIS', 'Debit'];
    var now = new Date();
    var dateStr = Utilities.formatDate(now, Session.getScriptTimeZone(), 'yyyyMMdd');

    for (var s = 0; s < 5; s++) {
      var itemCount = Math.floor(Math.random() * 3) + 1;
      var items = [];
      var subtotal = 0;

      for (var k = 0; k < itemCount; k++) {
        var mi = menuItems[Math.floor(Math.random() * menuItems.length)];
        var qty = Math.floor(Math.random() * 2) + 1;
        items.push({ ItemID: mi.ItemID, ItemName: mi.ItemName, Quantity: qty, UnitPrice: mi.Price });
        subtotal += mi.Price * qty;
      }

      var total = subtotal;
      var trxTime = new Date(now.getTime() - (s * 60 * 60 * 1000));
      var trxId = 'INV-' + dateStr + '-SMP' + String(s + 1).padStart(3, '0');

      sheet.appendRow([
        trxId,
        trxTime,
        cashiers[s % cashiers.length],
        JSON.stringify(items),
        subtotal,
        0,
        0,
        total,
        paymentMethods[s % paymentMethods.length],
        total,
        0,
        'Completed'
      ]);
    }

    return { success: true, message: '5 transaksi sample berhasil dibuat' };
  } catch (error) {
    return { success: false, message: error.message };
  }
}

function processTransaction(transactionData) {
  try {
    var lock = LockService.getScriptLock();
    lock.waitLock(30000);
    
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('Transactions');
    if (!sheet) {
      sheet = ss.insertSheet('Transactions');
      sheet.appendRow(['TransactionID', 'Timestamp', 'CashierName', 'Items', 
                       'Subtotal', 'Discount', 'Tax', 'Total', 
                       'PaymentMethod', 'AmountPaid', 'Change', 'Status']);
    }
    
    var date = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyyMMdd');
    var count = sheet.getLastRow();
    var seq = String(count).padStart(4, '0');
    var trxId = 'INV-' + date + '-' + seq;
    
    var items = transactionData.items || [];
    var subtotal = transactionData.subtotal || 0;
    var discount = transactionData.discount || 0;
    var tax = transactionData.tax || 0;
    var total = subtotal - discount + tax;
    
    sheet.appendRow([
      trxId,
      new Date(),
      transactionData.cashierName || 'Admin',
      JSON.stringify(items),
      subtotal,
      discount,
      tax,
      total,
      transactionData.paymentMethod || 'Cash',
      transactionData.amountPaid || total,
      (transactionData.amountPaid || total) - total,
      'Completed'
    ]);
    
    try {
      var stockSheet = ss.getSheetByName('Inventory_Stock');
      if (stockSheet && stockSheet.getLastRow() > 1) {
        var stockData = stockSheet.getDataRange().getValues();
        for (var i = 0; i < items.length; i++) {
          var itemId = items[i].ItemID;
          var qty = items[i].Quantity || 1;
          for (var j = 1; j < stockData.length; j++) {
            if (String(stockData[j][0]) === String(itemId)) {
              var currentStock = parseFloat(stockData[j][1]) || 0;
              var newStock = Math.max(0, currentStock - qty);
              stockSheet.getRange(j + 1, 2).setValue(newStock);
              break;
            }
          }
        }
      }
    } catch (stockError) {}
    
    lock.releaseLock();
    
    return {
      success: true,
      transactionId: trxId,
      total: total
    };
  } catch (error) {
    return { success: false, message: error.message };
  }
}

function getDashboardData() {
  try {
    var allTransactions = getTransactionHistory(1000);
    var today = new Date();
    var todayStr = Utilities.formatDate(today, Session.getScriptTimeZone(), 'yyyy-MM-dd');
    
    var revenueToday = 0;
    var transactionsToday = 0;
    var totalRevenue = 0;
    var totalTransactions = allTransactions.length;
    var totalCOGS = 0;
    
    var menuItems = getMenuItems();
    var costMap = {};
    for (var i = 0; i < menuItems.length; i++) {
      costMap[menuItems[i].ItemID] = menuItems[i].Cost || 0;
    }
    
    for (var i = 0; i < allTransactions.length; i++) {
      var trx = allTransactions[i];
      totalRevenue += trx.Total || 0;
      
      var trxDate = new Date(trx.Timestamp);
      var trxDateStr = Utilities.formatDate(trxDate, Session.getScriptTimeZone(), 'yyyy-MM-dd');
      if (trxDateStr === todayStr) {
        revenueToday += trx.Total || 0;
        transactionsToday++;
      }
      
      var items = trx.Items || [];
      for (var j = 0; j < items.length; j++) {
        var item = items[j];
        var cost = costMap[item.ItemID] || 0;
        totalCOGS += cost * (item.Quantity || 1);
      }
    }
    
    var grossProfit = totalRevenue - totalCOGS;
    var profitMargin = totalRevenue > 0 ? (grossProfit / totalRevenue) * 100 : 0;
    
    var stockSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Inventory_Stock');
    var lowStock = 0;
    if (stockSheet && stockSheet.getLastRow() > 1) {
      var data = stockSheet.getDataRange().getValues();
      for (var i = 1; i < data.length; i++) {
        var row = data[i];
        if (row) {
          var current = parseFloat(row[1]) || 0;
          var min = parseFloat(row[2]) || 0;
          if (current <= min && current > 0) lowStock++;
        }
      }
    }
    
    return {
      revenueToday: revenueToday,
      transactionsToday: transactionsToday,
      totalRevenue: totalRevenue,
      totalTransactions: totalTransactions,
      totalCOGS: totalCOGS,
      grossProfit: grossProfit,
      profitMargin: profitMargin,
      lowStock: lowStock
    };
  } catch (error) {
    return { 
      revenueToday: 0, 
      transactionsToday: 0, 
      totalRevenue: 0, 
      totalTransactions: 0,
      totalCOGS: 0,
      grossProfit: 0,
      profitMargin: 0,
      lowStock: 0 
    };
  }
}

function getWeeklySalesData() {
  try {
    var allTransactions = getTransactionHistory(1000);
    var today = new Date();
    var weekData = [];
    var days = ['Min', 'Sen', 'Sel', 'Rab', 'Kam', 'Jum', 'Sab'];
    
    for (var i = 6; i >= 0; i--) {
      var date = new Date(today);
      date.setDate(date.getDate() - i);
      var dateStr = Utilities.formatDate(date, Session.getScriptTimeZone(), 'yyyy-MM-dd');
      weekData.push({
        date: dateStr,
        day: days[date.getDay()],
        revenue: 0,
        transactions: 0,
        profit: 0,
        cogs: 0
      });
    }
    
    var menuItems = getMenuItems();
    var costMap = {};
    for (var i = 0; i < menuItems.length; i++) {
      costMap[menuItems[i].ItemID] = menuItems[i].Cost || 0;
    }
    
    for (var i = 0; i < allTransactions.length; i++) {
      var trx = allTransactions[i];
      var trxDate = new Date(trx.Timestamp);
      var trxDateStr = Utilities.formatDate(trxDate, Session.getScriptTimeZone(), 'yyyy-MM-dd');
      
      for (var j = 0; j < weekData.length; j++) {
        if (weekData[j].date === trxDateStr) {
          weekData[j].revenue += trx.Total || 0;
          weekData[j].transactions++;
          
          var items = trx.Items || [];
          for (var k = 0; k < items.length; k++) {
            var item = items[k];
            var cost = costMap[item.ItemID] || 0;
            weekData[j].cogs += cost * (item.Quantity || 1);
          }
          break;
        }
      }
    }
    
    for (var j = 0; j < weekData.length; j++) {
      weekData[j].profit = weekData[j].revenue - weekData[j].cogs;
    }
    
    return weekData;
  } catch (error) {
    return [];
  }
}

function getProfitLossReport(startDate, endDate) {
  try {
    var allTransactions = getTransactionHistory(1000);
    var menuItems = getMenuItems();
    
    var costMap = {};
    for (var i = 0; i < menuItems.length; i++) {
      costMap[menuItems[i].ItemID] = menuItems[i].Cost || 0;
    }
    
    var start = new Date(startDate);
    var end = new Date(endDate);
    end.setHours(23, 59, 59, 999);
    
    var summary = {
      totalRevenue: 0,
      totalCOGS: 0,
      totalGrossProfit: 0,
      totalTransactions: 0,
      totalDiscount: 0,
      totalTax: 0,
      netProfit: 0,
      profitMargin: 0,
      items: {},
      categories: {}
    };
    
    var itemDetails = {};
    
    for (var i = 0; i < allTransactions.length; i++) {
      var trx = allTransactions[i];
      var trxDate = new Date(trx.Timestamp);
      
      if (trxDate >= start && trxDate <= end) {
        summary.totalRevenue += trx.Total || 0;
        summary.totalDiscount += trx.Discount || 0;
        summary.totalTax += trx.Tax || 0;
        summary.totalTransactions++;
        
        var items = trx.Items || [];
        for (var j = 0; j < items.length; j++) {
          var item = items[j];
          var itemId = item.ItemID || '';
          var itemName = item.ItemName || 'Unknown';
          var qty = item.Quantity || 1;
          var unitPrice = item.UnitPrice || 0;
          var cost = costMap[itemId] || 0;
          var totalPrice = unitPrice * qty;
          var totalCost = cost * qty;
          var profit = totalPrice - totalCost;
          
          if (!itemDetails[itemName]) {
            itemDetails[itemName] = {
              name: itemName,
              id: itemId,
              category: '',
              quantity: 0,
              revenue: 0,
              cost: 0,
              profit: 0,
              margin: 0
            };
            
            for (var k = 0; k < menuItems.length; k++) {
              if (menuItems[k].ItemID === itemId) {
                itemDetails[itemName].category = menuItems[k].Category || '';
                break;
              }
            }
          }
          
          itemDetails[itemName].quantity += qty;
          itemDetails[itemName].revenue += totalPrice;
          itemDetails[itemName].cost += totalCost;
          itemDetails[itemName].profit += profit;
          
          summary.totalCOGS += totalCost;
        }
      }
    }
    
    var itemNames = Object.keys(itemDetails);
    for (var i = 0; i < itemNames.length; i++) {
      var item = itemDetails[itemNames[i]];
      item.margin = item.revenue > 0 ? (item.profit / item.revenue) * 100 : 0;
    }
    
    var categoryDetails = {};
    for (var i = 0; i < itemNames.length; i++) {
      var item = itemDetails[itemNames[i]];
      var cat = item.category || 'Other';
      if (!categoryDetails[cat]) {
        categoryDetails[cat] = {
          name: cat,
          revenue: 0,
          cost: 0,
          profit: 0,
          quantity: 0,
          margin: 0
        };
      }
      categoryDetails[cat].revenue += item.revenue;
      categoryDetails[cat].cost += item.cost;
      categoryDetails[cat].profit += item.profit;
      categoryDetails[cat].quantity += item.quantity;
    }
    
    var catNames = Object.keys(categoryDetails);
    for (var i = 0; i < catNames.length; i++) {
      var cat = categoryDetails[catNames[i]];
      cat.margin = cat.revenue > 0 ? (cat.profit / cat.revenue) * 100 : 0;
    }
    
    summary.totalGrossProfit = summary.totalRevenue - summary.totalCOGS;
    summary.netProfit = summary.totalGrossProfit - summary.totalDiscount - summary.totalTax;
    summary.profitMargin = summary.totalRevenue > 0 ? (summary.netProfit / summary.totalRevenue) * 100 : 0;
    summary.items = itemDetails;
    summary.categories = categoryDetails;
    
    return summary;
  } catch (error) {
    return { error: error.message };
  }
}

function getShiftReport(startDate, endDate, cashierName) {
  try {
    var allTransactions = getTransactionHistory(1000);
    var start = new Date(startDate);
    var end = new Date(endDate);
    end.setHours(23, 59, 59, 999);
    
    var report = {
      cashierName: cashierName || 'All',
      totalTransactions: 0,
      totalRevenue: 0,
      totalDiscount: 0,
      totalTax: 0,
      netRevenue: 0,
      averageTransaction: 0,
      paymentMethods: {},
      transactions: []
    };
    
    for (var i = 0; i < allTransactions.length; i++) {
      var trx = allTransactions[i];
      var trxDate = new Date(trx.Timestamp);
      
      if (trxDate >= start && trxDate <= end) {
        if (cashierName && trx.CashierName !== cashierName) continue;
        
        report.totalTransactions++;
        report.totalRevenue += trx.Total || 0;
        report.totalDiscount += trx.Discount || 0;
        report.totalTax += trx.Tax || 0;
        
        var method = trx.PaymentMethod || 'Cash';
        if (!report.paymentMethods[method]) {
          report.paymentMethods[method] = { count: 0, amount: 0 };
        }
        report.paymentMethods[method].count++;
        report.paymentMethods[method].amount += trx.Total || 0;
        
        report.transactions.push({
          id: trx.TransactionID,
          time: trx.Timestamp,
          total: trx.Total,
          method: method
        });
      }
    }
    
    report.netRevenue = report.totalRevenue - report.totalDiscount - report.totalTax;
    report.averageTransaction = report.totalTransactions > 0 ? report.totalRevenue / report.totalTransactions : 0;
    
    return report;
  } catch (error) {
    return { error: error.message };
  }
}

        };
      }

function validateSession(sessionId) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('User_Sessions');
    if (!sheet) return null;
    
    var data = sheet.getDataRange().getValues();
    for (var i = data.length - 1; i >= 1; i--) {
      var row = data[i];
      if (row[0] === sessionId) {
        var sessionData = JSON.parse(row[1]);
        var lastActivity = new Date(row[2]);
        var now = new Date();
        var diffMinutes = (now - lastActivity) / (1000 * 60);
        
        if (diffMinutes > 480) {
          sheet.deleteRow(i + 1);
          return null;
        }
        
        sheet.getRange(i + 1, 3).setValue(now);
        return sessionData;
      }
    }
    return null;
  } catch (error) {
    return null;
  }
}

function generateSessionId() {
  return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, function(c) {
    var r = Math.random() * 16 | 0, v = c == 'x' ? r : (r & 0x3 | 0x8);
    return v.toString(16);
  });
}

function sendLowStockAlert() {
  try {
    var inventory = getInventoryItems();
    var menuItems = getMenuItems();
    var lowItems = [];
    
    var nameMap = {};
    for (var i = 0; i < menuItems.length; i++) {
      nameMap[menuItems[i].ItemID] = menuItems[i].ItemName;
    }
    
    for (var i = 0; i < inventory.length; i++) {
      var item = inventory[i];
      if (item.CurrentStock <= item.MinStock && item.CurrentStock > 0) {
        lowItems.push({
          id: item.ItemID,
          name: nameMap[item.ItemID] || item.ItemID,
          stock: item.CurrentStock,
          min: item.MinStock,
          unit: item.Unit || ''
        });
      }
    }
    
    if (lowItems.length === 0) {
      return { success: true, message: 'Tidak ada stok menipis' };
    }
    
    var subject = '⚠️ ALERT: Stok Menipis - Menteri Cafe';
    var body = 'Yth. Owner/Manager,\n\n';
    body += 'Berikut adalah daftar item yang stoknya menipis:\n\n';
    body += '='.repeat(40) + '\n';
    for (var i = 0; i < lowItems.length; i++) {
      var item = lowItems[i];
      body += item.id + ' | ' + item.name + ' | ' + item.stock + ' | ' + item.min + ' | ' + item.unit + '\n';
    }
    body += '='.repeat(40) + '\n\n';
    body += 'Total ' + lowItems.length + ' item perlu segera di-restock.\n\n';
    body += 'Terima kasih,\nMenteri Cafe System';
    
    MailApp.sendEmail('owner@mentericafe.com', subject, body);
    
    return { success: true, message: 'Alert sent', items: lowItems.length };
  } catch (error) {
    return { success: false, message: error.message };
  }
}

function sendDailyReportEmail() {
  try {
    var yesterday = new Date();
    yesterday.setDate(yesterday.getDate() - 1);
    var dateStr = Utilities.formatDate(yesterday, Session.getScriptTimeZone(), 'yyyy-MM-dd');
    
    var allTransactions = getTransactionHistory(1000);
    var revenue = 0;
    var count = 0;
    for (var i = 0; i < allTransactions.length; i++) {
      var trxDate = new Date(allTransactions[i].Timestamp);
      var trxDateStr = Utilities.formatDate(trxDate, Session.getScriptTimeZone(), 'yyyy-MM-dd');
      if (trxDateStr === dateStr) {
        revenue += allTransactions[i].Total || 0;
        count++;
      }
    }
    
    var subject = '📊 Daily Report - Menteri Cafe ' + dateStr;
    var body = 'Yth. Owner/Manager,\n\n';
    body += 'Berikut adalah laporan harian:\n\n';
    body += '📅 Tanggal: ' + dateStr + '\n';
    body += '💰 Total Revenue: Rp ' + formatRupiah(revenue) + '\n';
    body += '📋 Total Transaksi: ' + count + '\n';
    body += '\nTerima kasih,\nMenteri Cafe System';
    
    MailApp.sendEmail('owner@mentericafe.com', subject, body);
    
    return { success: true };
  } catch (error) {
    return { success: false, message: error.message };
  }
}

function formatRupiah(number) {
  if (number === undefined || number === null || isNaN(number)) number = 0;
  return Math.round(number).toString().replace(/\B(?=(\d{3})+(?!\d))/g, '.');
}

function escapeHtml(str) {
  if (str === undefined || str === null) return '';
  return String(str)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');
}
